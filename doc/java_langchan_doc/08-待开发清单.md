```java
/**
 * 把一条落库消息还原成接口要的结构；tool_calls 是 JSON 字符串，这里解析成对象列表。
 *
 * <p>解析失败时按「无轨迹」处理，绝不让一条脏数据打断整个历史回放。</p>
 *
 * @param row 消息实体
 * @return 接口消息结构
 */
private ChatDtos.StoredChatMessage toStoredMessage(ChatMessage row) {
    List<ChatDtos.ToolCallTrace> toolCalls = null;
    // tool_calls 只在助手消息里有值；用户消息是 null，直接跳过解析
    if (row.getToolCalls() != null && !row.getToolCalls().isBlank()) {
        try {
            toolCalls = objectMapper.readValue(row.getToolCalls(),
                    new TypeReference<List<ChatDtos.ToolCallTrace>>() { });
        } catch (Exception e) {
            // 历史里若存过旧格式，解析不动就丢弃轨迹，不影响正文回放
            toolCalls = null;
        }
    }
    return new ChatDtos.StoredChatMessage(
            row.getId() == null ? 0L : row.getId(),
            row.getRole(),
            row.getContent(),
            row.getMode(),
            toolCalls,
            row.getFinishReason(),
            row.getElapsed(),
            row.getCreatedAt());
}

/**
 * 把 ISO-8601 时间字符串转成 Unix 秒（浮点）。
 *
 * <p>SessionInfo 的 created_at / last_active 是「秒」；格式对不上时回 null，前端按「未知」展示。</p>
 *
 * @param iso 形如 2026-10-27T16:30:00 的字符串
 * @return Unix 秒；无法解析时为 null
 */
private Double parseEpochSeconds(String iso) {
    if (iso == null || iso.isBlank()) {
        return null;
    }
    try {
        return Instant.parse(iso).toEpochMilli() / 1000.0;
    } catch (Exception e) {
        return null;
    }
}

/**
 * 取会话标题，导出 Markdown 的标题行用。
 *
 * @param sessionId 会话 id
 * @return 标题；会话不存在时回「会话」
 */
private String sessionTitle(String sessionId) {
    return chatSessionRepository.findById(sessionId)
            .map(ChatSession::getTitle)
            .orElse("会话");
}
```

### 1.1 `history()` —— 历史回放（`GET /chat/history/sessions/{id}`）

```java
/**
 * 读取某个会话的历史消息（最近 limit 条，时间正序返回）。
 *
 * <p>limit 控制「往回看多少条」，返回顺序仍保持旧→新，前端按数组顺序渲染即可。</p>
 *
 * @param sessionId 会话 id
 * @param limit     最多返回多少条
 * @return 历史消息
 */
@Override
public ChatDtos.ChatHistory history(String sessionId, int limit) {
    int safeLimit = limit <= 0 ? 20 : Math.min(limit, 100);
    List<ChatMessage> rows = chatPersistence.findMessages(sessionId);
    // 取「最近 safeLimit 条」：仓库按 id 升序返回，所以从尾部切片，保持时间正序
    int from = Math.max(0, rows.size() - safeLimit);
    List<ChatDtos.StoredChatMessage> messages = new ArrayList<>();
    for (int i = from; i < rows.size(); i++) {
        messages.add(toStoredMessage(rows.get(i)));
    }
    return new ChatDtos.ChatHistory(sessionId, messages);
}
```

### 1.2 `patchSession()` —— 改标题/置顶/归档（`PATCH /chat/history/sessions/{id}`）

```java
/**
 * 修改会话标题/置顶/归档；只改「传了」的字段。
 *
 * <p>标题的约定：null 或空串都视为「不改」，避免前端误发空标题把标题清掉。</p>
 *
 * @param sessionId 会话 id
 * @param patch     要修改的字段
 * @return 落库后的标记
 */
@Override
public ChatDtos.ChatSessionPatchResult patchSession(String sessionId, ChatDtos.ChatSessionPatch patch) {
    ChatSession row = chatPersistence.findSession(sessionId).orElse(null);
    // 会话不存在：回全空，前端据此提示「会话已不在」
    if (row == null) {
        return new ChatDtos.ChatSessionPatchResult(false, sessionId, "", false, false);
    }
    boolean renamed = false;
    // 只有传了非空标题、且与旧标题不同才算「真的改名」
    if (patch.title() != null && !patch.title().isBlank() && !patch.title().equals(row.getTitle())) {
        row.setTitle(patch.title());
        renamed = true;
    }
    // 置顶/归档用包装类型 Boolean，null 表示「没传、不改」
    if (patch.pinned() != null) {
        row.setPinned(patch.pinned());
    }
    if (patch.archived() != null) {
        row.setArchived(patch.archived());
    }
    chatSessionRepository.save(row);
    return new ChatDtos.ChatSessionPatchResult(
            renamed, sessionId, row.getTitle(),
            Boolean.TRUE.equals(row.getPinned()), Boolean.TRUE.equals(row.getArchived()));
}
```

### 1.3 `export()` —— 导出（`GET .../export?format=md|json`）

```java
/**
 * 导出会话内容；md 给人读，json 给再加工。
 *
 * @param sessionId 会话 id
 * @param format    md 或 json（不区分大小写）；其它值回落到 md
 * @return 导出结果（含建议文件名与全文）
 */
@Override
public ChatDtos.ExportedSession export(String sessionId, String format) {
    List<ChatMessage> rows = chatPersistence.findMessages(sessionId);
    String fmt = "json".equalsIgnoreCase(format) ? "json" : "md";
    String content;
    if ("json".equals(fmt)) {
        // JSON 导出：直接序列化还原后的消息结构，方便程序再加工
        try {
            content = objectMapper.writerWithDefaultPrettyPrinter()
                    .writeValueAsString(rows.stream().map(this::toStoredMessage).toList());
        } catch (Exception e) {
            throw new ApiException(500, "序列化会话失败：" + e.getMessage());
        }
    } else {
        // Markdown 导出：标题 + 每条消息，适合人阅读
        StringBuilder sb = new StringBuilder();
        sb.append("# ").append(sessionTitle(sessionId)).append("\n\n");
        for (ChatMessage row : rows) {
            sb.append("## ").append("user".equals(row.getRole()) ? "用户" : "助手").append("\n\n");
            sb.append(row.getContent()).append("\n\n");
        }
        content = sb.toString();
    }
    return new ChatDtos.ExportedSession(sessionId, fmt, "chat-" + sessionId + "." + fmt, content, rows.size());
}
```

### 1.5 `deleteHistory()` —— 永久删除（`DELETE /chat/history/sessions/{id}`）

```java
/**
 * 永久删除某个会话及其全部消息。
 *
 * <p>顺序：先删消息、再删会话；最后清掉运行时状态，避免侧边栏还显示「运行中」。</p>
 *
 * @param sessionId 会话 id
 * @return 删除结果
 */
@Override
public ChatDtos.DeleteSessionResult deleteHistory(String sessionId) {
    boolean existed = chatPersistence.findSession(sessionId).isPresent();
    // 仓库没有按会话删除的派生方法，就把消息查出来整体删（会话级消息量不大，够用）
    List<ChatMessage> rows = chatMessageRepository.findBySessionIdOrderByIdAsc(sessionId);
    if (!rows.isEmpty()) {
        chatMessageRepository.deleteAll(rows);
    }
    if (existed) {
        chatSessionRepository.deleteById(sessionId);
    }
    // 清运行时状态（总线里的 Turn），让「已删会话」不再出现在运行态里
    sessionBus.finish(sessionId);
    return new ChatDtos.DeleteSessionResult(existed, sessionId);
}
```

### 2.0 完整替换版（含 import）

```java
// 模块说明：执行轨迹查询服务实现。数据源是 trace_event 表，把事件按「会话 → 轮次 → 事件」三级拼给前端。
package com.example.backendjava.service.impl;

// 本项目：轨迹实体
import com.example.backendjava.entity.TraceEvent;
// 本项目：轨迹仓库（已提供聚合与按会话查询）
import com.example.backendjava.repository.TraceEventRepository;
// 本项目：轨迹服务接口（本类实现它）
import com.example.backendjava.service.TraceService;
// Jackson：payload 是 JSON 字符串，还原成对象给前端
import com.fasterxml.jackson.databind.ObjectMapper;
// Jakarta 依赖注入：按类型注入字段
import jakarta.annotation.Resource;
// Spring Data：listSessionSummaries 的分页参数
import org.springframework.data.domain.PageRequest;
// Spring：声明为服务组件
import org.springframework.stereotype.Service;

// 标准库：ArrayList 收集列表
import java.util.ArrayList;
// 标准库：LinkedHashMap 保持返回字段顺序
import java.util.LinkedHashMap;
// 标准库：List / Map 返回值类型
import java.util.List;
import java.util.Map;

/**
 * 执行轨迹查询服务实现。
 */
@Service
public class TraceServiceImpl implements TraceService {

    /** 轨迹仓库。 */
    @Resource
    private TraceEventRepository repository;

    /** JSON 反序列化（payload 存的是字符串，返回给前端要还原成对象）。 */
    @Resource
    private ObjectMapper objectMapper;

    /** {@inheritDoc} */
    @Override
    public Map<String, Object> listSessions(int limit) {
        int safeLimit = limit <= 0 ? 50 : Math.min(limit, 200);
        // 聚合查询：一条 SQL 返回 (session_id, 事件条数, 最近时间)，按最近时间倒序
        List<Object[]> rows = repository.listSessionSummaries(PageRequest.of(0, safeLimit));
        List<Map<String, Object>> sessions = new ArrayList<>();
        for (Object[] row : rows) {
            String sessionId = String.valueOf(row[0]);
            Map<String, Object> item = new LinkedHashMap<>();
            item.put("session_id", sessionId);
            item.put("event_count", ((Number) row[1]).longValue());
            item.put("last_at", row[2]);
            // 轮次数与起始时间要逐条扫一遍，单独算
            item.put("turn_count", turnCount(sessionId));
            item.put("started_at", firstAt(sessionId));
            sessions.add(item);
        }
        return Map.of("sessions", sessions);
    }

    /** {@inheritDoc} */
    @Override
    public Map<String, Object> listTurns(String sessionId, int limit) {
        List<TraceEvent> rows = repository.findBySessionIdOrderByIdAsc(sessionId);
        // 按 turn_id 分组，且保持首次出现顺序（LinkedHashMap 保证）
        Map<String, List<TraceEvent>> byTurn = new LinkedHashMap<>();
        for (TraceEvent row : rows) {
            String turnId = row.getTurnId() == null ? "" : row.getTurnId();
            byTurn.computeIfAbsent(turnId, key -> new ArrayList<>()).add(row);
        }
        int safeLimit = limit <= 0 ? 50 : Math.min(limit, 200);
        List<Map<String, Object>> turns = new ArrayList<>();
        int i = 0;
        for (Map.Entry<String, List<TraceEvent>> entry : byTurn.entrySet()) {
            if (i++ >= safeLimit) {
                break;
            }
            List<TraceEvent> events = entry.getValue();
            Map<String, Object> turn = new LinkedHashMap<>();
            turn.put("turn_id", entry.getKey());
            turn.put("event_count", events.size());
            turn.put("started_at", events.get(0).getCreatedAt());
            turn.put("last_at", events.get(events.size() - 1).getCreatedAt());
            turns.add(turn);
        }
        Map<String, Object> result = new LinkedHashMap<>();
        result.put("session_id", sessionId);
        result.put("count", turns.size());
        result.put("turns", turns);
        return result;
    }

    /** {@inheritDoc} */
    @Override
    public Map<String, Object> listEvents(String sessionId, String turnId, int limit) {
        List<TraceEvent> rows = repository.findBySessionIdOrderByIdAsc(sessionId);
        // turn_id 为空表示整个会话；否则只保留该轮
        List<TraceEvent> filtered = rows.stream()
                .filter(row -> turnId == null || turnId.isBlank() || turnId.equals(row.getTurnId()))
                .toList();
        int safeLimit = limit <= 0 ? 500 : Math.min(limit, 2000);
        // 取最近 safeLimit 条，但保持时间正序
        int from = Math.max(0, filtered.size() - safeLimit);
        List<Map<String, Object>> events = new ArrayList<>();
        for (int i = from; i < filtered.size(); i++) {
            events.add(toEventMap(filtered.get(i)));
        }
        Map<String, Object> result = new LinkedHashMap<>();
        result.put("session_id", sessionId);
        result.put("turn_id", turnId);
        result.put("count", events.size());
        result.put("events", events);
        return result;
    }

    /** 一个会话的轮次数：按 turn_id 去重计数。 */
    private long turnCount(String sessionId) {
        return repository.findBySessionIdOrderByIdAsc(sessionId).stream()
                .map(TraceEvent::getTurnId)
                .distinct()
                .count();
    }

    /** 一个会话的第一条事件时间；没有则为 null。 */
    private String firstAt(String sessionId) {
        List<TraceEvent> rows = repository.findBySessionIdOrderByIdAsc(sessionId);
        return rows.isEmpty() ? null : rows.get(0).getCreatedAt();
    }

    /** 轨迹实体 → 字典；payload 解析成对象，前端直接读。 */
    private Map<String, Object> toEventMap(TraceEvent row) {
        Map<String, Object> map = new LinkedHashMap<>();
        map.put("id", row.getId());
        map.put("session_id", row.getSessionId());
        map.put("turn_id", row.getTurnId());
        map.put("type", row.getType());
        try {
            map.put("payload", objectMapper.readValue(row.getPayload(), Map.class));
        } catch (Exception e) {
            // payload 存的是 JSON 字符串，解析不动就回空对象，不打断整页
            map.put("payload", Map.of());
        }
        map.put("created_at", row.getCreatedAt());
        return map;
    }
}
```

```java
// 模块说明：Ollama 模型工厂。本地部署、不需要 API Key，是三个待补工厂里最容易验证的一个。
package com.example.backendjava.service.impl;

// Spring 组件注解：把本工厂注册成 Bean，由路由器自动发现
import org.springframework.stereotype.Component;
// LangChain4j 同步聊天模型（Ollama 实现）
import dev.langchain4j.model.ollama.OllamaChatModel;
// LangChain4j 流式聊天模型（Ollama 实现）
import dev.langchain4j.model.ollama.OllamaStreamingChatModel;
// LangChain4j 同步模型接口：方法返回值类型
import dev.langchain4j.model.chat.ChatModel;
// LangChain4j 流式模型接口：方法返回值类型
import dev.langchain4j.model.chat.StreamingChatModel;
// 项目内业务异常：缺 baseUrl 时给可读提示
import com.example.backendjava.common.ApiException;

// 模型工厂扩展点：按供应商装配 LangChain4j ChatModel
import com.example.backendjava.llm.LlmModelFactory;
// 装配一个模型实例所需的参数
import com.example.backendjava.dto.LlmModelSpec;

/**
 * Ollama 模型工厂。
 *
 * <p>Ollama 本地起服务后提供 OpenAI 兼容接口，LangChain4j 有专门实现；
 * 不需要密钥，所以不校验 apiKey，只校验 baseUrl 与 modelName。</p>
 */
@Component
public class OllamaLlmModelFactory implements LlmModelFactory {

    /**
     * 支持的供应商 id。
     *
     * @param providerId 供应商 id
     * @return 是否为 ollama
     */
    @Override
    public boolean supports(String providerId) {
        return "ollama".equals(providerId);
    }

    /**
     * 创建同步聊天模型。
     *
     * @param spec 模型装配参数
     * @return OllamaChatModel 实例
     * @throws ApiException baseUrl 或 model 缺失时抛出
     */
    @Override
    public ChatModel createChatModel(LlmModelSpec spec) {
        validate(spec);
        var builder = OllamaChatModel.builder()
                .baseUrl(spec.baseUrl())
                .modelName(spec.model());
        // 只有显式配置了温度才覆盖默认值
        if (spec.temperature() != null) {
            builder.temperature(spec.temperature());
        }
        return builder.build();
    }

    /**
     * 创建流式聊天模型。
     *
     * @param spec 模型装配参数
     * @return OllamaStreamingChatModel 实例
     * @throws ApiException baseUrl 或 model 缺失时抛出
     */
    @Override
    public StreamingChatModel createStreamingModel(LlmModelSpec spec) {
        validate(spec);
        var builder = OllamaStreamingChatModel.builder()
                .baseUrl(spec.baseUrl())
                .modelName(spec.model());
        if (spec.temperature() != null) {
            builder.temperature(spec.temperature());
        }
        return builder.build();
    }

    /**
     * 校验构建模型必需的配置。
     *
     * @param spec 模型装配参数
     * @throws ApiException baseUrl 或 model 为空时抛出 400
     */
    private void validate(LlmModelSpec spec) {
        if (spec.baseUrl() == null || spec.baseUrl().isBlank()) {
            throw new ApiException(400, "缺少 Ollama 接口地址 baseUrl");
        }
        if (spec.model() == null || spec.model().isBlank()) {
            throw new ApiException(400, "缺少模型名 model");
        }
    }
}
```

### 4.2 Anthropic / Gemini 同结构，只换类名与 builder

| 供应商 | 同步模型类 | 流式模型类 | builder 关键参数 |
| --- | --- | --- | --- |
| Anthropic | `dev.langchain4j.model.anthropic.AnthropicChatModel` | `AnthropicStreamingChatModel` | `.apiKey(spec.apiKey()).modelName(spec.model()).temperature(...)`；`supports` 判 `"anthropic"`，校验 apiKey |
| Gemini | `dev.langchain4j.model.googleai.GoogleAiGeminiChatModel` | `GoogleAiGeminiStreamingChatModel` | `.apiKey(spec.apiKey()).modelName(spec.model()).temperature(...)`；`supports` 判 `"google"`，校验 apiKey |

> 结构照抄 4.1 的 Ollama 工厂，三处不同：`supports` 的判定值、`builder` 用到的模型类、`validate` 里
> 多校验一句 `apiKey`（Anthropic/Gemini 都是云端、必须带 Key）。

### 4.3 `custom-*` 供应商（注意：不是新工厂）

`OpenAiCompatibleLlmModelFactory.supports()` **已经支持** `providerId.startsWith("custom-")`。
真正缺的是 `LlmProviderRegistry.listProviders()` 里的 TODO——把设置库 `llm.custom_providers` /
`llm.custom_models` 读出来追加到 `BUILTIN_PROVIDERS`，并给 id 加 `custom-` 前缀。这是注册表改动，不是工厂。
